% Title format for parts and chapters
\titleformat{\part}[display]
    {\normalfont\sffamily\Huge\bfseries\color{primary}\centering\hyphenpenalty=10000\exhyphenpenalty=10000}
    {\LARGE\color{secondary}\partname\ \thepart}
    {2ex}
    {\titlerule[2pt]\vspace{2ex}}
    [\vspace{2ex}\titlerule]

\titleformat{\chapter}[display]
    {\normalfont\sffamily\huge\bfseries\color{primary}\raggedright\hyphenpenalty=10000\exhyphenpenalty=10000}

    {1ex}
    {\titlerule[1.5pt]\vspace{2ex}\raggedright}
    [\vspace{1ex}\titlerule]
